25constexpr bool needs_sum_check() {

36template<typename Dest_, typename First_, typename Second_>
37constexpr bool does_sum_overflow_ordered(First_ first, Second_ second) {
38 constexpr Dest_ dest_maxed = get_max<Dest_>();
39 const auto first_val = get_value(first);
40 if (is_greater_than(first_val, dest_maxed)) { // This check can be compiled out if First_ is not greater than Dest_.
41 return true;
42 } else {
43 const Dest_ delta = dest_maxed - static_cast<Dest_>(first_val); // If First_ is Constant, then the compiler could compute 'delta' at compile time.
44 return is_greater_than(get_value(second), delta);
45 }

48template<typename Dest_, typename First_, typename Second_>
49constexpr bool does_sum_overflow(First_ first, Second_ second) {
50 if constexpr(is_Constant<First_>::value) {
51 // Shift threshold calculation to compile-time.
52 return does_sum_overflow_ordered<Dest_>(first, second);
53
54 } else if constexpr(is_Constant<Second_>::value) {
55 // Shift threshold calculation to compile-time.
56 return does_sum_overflow_ordered<Dest_>(second, first);
57
58 } else if constexpr(is_less_than_or_equal(get_max<First_>(), get_max<Dest_>())) {
59 // Skip overflow check when First_ is not greater than Dest_.
60 return does_sum_overflow_ordered<Dest_>(first, second);
61
62 } else {
63 // Skip overflow check if Second_ is not greater than Dest_.
64 // If not, the order doesn't matter w.r.t. the amount of work, so we might as well use this as the default case.
65 return does_sum_overflow_ordered<Dest_>(second, first);
66 }

69template<typename Dest_, typename First_, typename Second_>
70constexpr auto sum_protected(First_ first, Second_ second) {
71 static_assert(std::is_integral<Dest_>::value);
72
73 if constexpr(needs_sum_check<Dest_, First_, Second_>()) {
74 if (does_sum_overflow<Dest_>(first, second)) {
75 return std::optional<Dest_>();
76 } else {
77 // Cast before sum to avoid overflow in the input types or whatever integer promotion decides to use.
78 return std::make_optional<Dest_>(static_cast<Dest_>(get_value(first)) + static_cast<Dest_>(get_value(second)));
79 }
80
81 } else {
82 constexpr Dest_ maxsum = static_cast<Dest_>(get_max<First_>()) + static_cast<Dest_>(get_max<Second_>());
83 if constexpr(is_Constant<First_>::value && is_Constant<Second_>::value) {
84 // Return a Constant in case we can do more compile-time stuff in the next chained operation.
85 // No need to this when 'needs_sum_check' is true as it will always fail with two Constants.
86 return std::make_optional(Constant<Dest_, maxsum>());
87 } else {
88 return std::make_optional<Bounded<Dest_, maxsum> >(static_cast<Dest_>(get_value(first)) + static_cast<Dest_>(get_value(second)));
89 }
90 }
91}
