24template<typename Dest_, typename First_, typename Second_>
25constexpr bool needs_product_check() {
26 constexpr Dest_ dest_maxed = get_max<Dest_>();
27 constexpr auto first_maxed = get_max<First_>();
28 if constexpr(is_greater_than(first_maxed, dest_maxed)) {
29 return true;
30 } else if constexpr(first_maxed == 0) {
31 // We are obliged to check that all arguments fit into Dest_, but this is only necessary if first_maxed == 0.
32 return is_greater_than(get_max<Second_>(), dest_maxed);
33 } else {
34 constexpr auto ratio = dest_maxed / static_cast<Dest_>(first_maxed);
35 return is_greater_than(get_max<Second_>(), ratio);
36 }

39template<typename Dest_, typename First_, typename Second_>
40constexpr bool does_product_overflow_ordered(First_ first, Second_ second) {
41 constexpr Dest_ dest_maxed = get_max<Dest_>();
42 const auto first_val = get_value(first);
43 if (is_greater_than(first_val, dest_maxed)) { // This check can be compiled out if First_ is not greater than Dest_.
44 return true;
45 } else if (first_val == 0) {
46 // We are obliged to check that all arguments fit into Dest_, but this is only necessary if first_val == 0.
47 return is_greater_than(get_value(second), dest_maxed);
48 } else {
49 const auto ratio = dest_maxed / static_cast<Dest_>(first_val); // If First_ is Constant, then the compiler could compute 'ratio' at compile time.
50 return is_greater_than(get_value(second), ratio);
51 }

54template<typename Dest_, typename First_, typename Second_>
55constexpr bool does_product_overflow(First_ first, Second_ second) {
56 if constexpr(is_Constant<First_>::value) {
57 // Shift threshold calculation to compile-time.
58 return does_product_overflow_ordered<Dest_>(first, second);
59
60 } else if constexpr(is_Constant<Second_>::value) {
61 // Shift threshold calculation to compile-time.
62 return does_product_overflow_ordered<Dest_>(second, first);
63
64 } else if constexpr(is_less_than_or_equal(get_max<First_>(), get_max<Dest_>())) {
65 // Skip overflow check when First_ is not greater than Dest_.
66 return does_product_overflow_ordered<Dest_>(first, second);
67
68 } else {
69 // Skip overflow check if Second_ is not greater than Dest_.
70 // If not, the order doesn't matter w.r.t. the amount of work, so we might as well use this as the default case.
71 return does_product_overflow_ordered<Dest_>(second, first);
72 }
73}

75template<typename Dest_, typename First_, typename Second_>
76constexpr auto product_protected(First_ first, Second_ second) {
77 static_assert(std::is_integral<Dest_>::value);
78
79 if constexpr(needs_product_check<Dest_, First_, Second_>()) {
80 if (does_product_overflow<Dest_>(first, second)) {
81 return std::optional<Dest_>();
82 } else {
83 // Cast before product to avoid overflow in the input types or whatever integer promotion decides to use.
84 return std::make_optional<Dest_>(static_cast<Dest_>(get_value(first)) * static_cast<Dest_>(get_value(second)));
85 }
86
87 } else {
88 constexpr Dest_ maxprod = static_cast<Dest_>(get_max<First_>()) * static_cast<Dest_>(get_max<Second_>());
89 if constexpr(is_Constant<First_>::value && is_Constant<Second_>::value) {
90 // Return a Constant in case we can do more compile-time stuff in the next chained operation.
91 // No need to this when 'needs_product_check' is true as it will always fail with two Constants.
92 return std::make_optional(Constant<Dest_, maxprod>());
93 } else {
94 return std::make_optional<Bounded<Dest_, maxprod> >(static_cast<Dest_>(get_value(first)) * static_cast<Dest_>(get_value(second)));
95 }
96 }
97}
